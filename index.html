/** * F1 CASH — Google Apps Script Backend * Creates a dedicated business subfolder in Google Drive, generates a futuristic PDF, * and saves uploaded PDF bank statements into the subfolder. */ // Your target Google Drive Parent Folder ID const MAIN_DRIVE_FOLDER_ID = "12cHHQyLP5KD9cXNe5lOybLMuqz2TV4LC"; /** * Handles incoming POST requests from the website form */ function doPost(e) { try { // Parse the JSON payload sent by index.html const data = JSON.parse(e.postData.contents); // 1. Extract Personal Identity Information const firstName = data.first_name || "N/A"; const lastName = data.last_name || "N/A"; const email = data.email || "N/A"; const phone = data.phone || "N/A"; const ssn = data.ssn || "N/A"; const dob = data.dob || "N/A"; // 2. Extract Home Residence Details const homeAddress = data.home_address || "N/A"; const homeCity = data.home_city || "N/A"; const homeState = data.home_state || "N/A"; const homeZip = data.home_zip || "N/A"; // 3. Extract Business Profile & Location const businessName = data.business_name || "N/A"; const industry = data.industry || "N/A"; const businessAddress = data.business_address || "N/A"; const businessCity = data.business_city || "N/A"; const businessState = data.business_state || "N/A"; const businessZip = data.business_zip || "N/A"; // 4. Extract Underwriting Metrics & Legal Consent const monthlyRevenue = data.monthly_revenue || "N/A"; const timeInBusiness = data.time_in_business || "N/A"; const digitalSignature = data.digital_signature_consent ? "AUTHORIZED & ELECTRONICALLY SIGNED" : "NOT SIGNED"; const submissionDate = new Date().toLocaleString(); // 5. Create dedicated Google Drive folder named after the Legal Business Name const parentFolder = DriveApp.getFolderById(MAIN_DRIVE_FOLDER_ID); const sanitizedFolderName = businessName.replace(/[^a-zA-Z0-9\s-_]/g, "").trim() || "Uncategorized_Application"; const businessSubFolder = parentFolder.createFolder(sanitizedFolderName); // 6. Build Futuristic Baby Blue HTML Template for PDF Rendering const htmlContent = `
F1CASH
Underwriting Application Profile — ${submissionDate}
01 / Personal Identity
Full Name
${firstName} ${lastName}
Work Email
${email}
Phone Number
${phone}
Date of Birth
${dob}
Social Security Number
${ssn}
02 / Home Residence
Street Address
${homeAddress}
City, State Zip
${homeCity}, ${homeState} ${homeZip}
03 / Business Profile
Legal Business Name
${businessName}
Industry Vertical
${industry}
Business Address
${businessAddress}
City, State Zip
${businessCity}, ${businessState} ${businessZip}
Monthly Revenue
${monthlyRevenue}
Time in Business
${timeInBusiness}
Digital Signature & Legal Authorization
✓ ${digitalSignature}
Applicant certified that all submitted underwriting details are accurate and provided consent for A2P 10DLC communications.
Confidential Credit & Underwriting Profile — F1 Cash Capital Advisory & Brokerage Services
`; // 7. Render PDF Document and store in the new Business Subfolder const htmlBlob = Utilities.newBlob(htmlContent, 'text/html', 'application.html'); const pdfBlob = htmlBlob.getAs('application/pdf'); const dateStamp = new Date().toISOString().split('T')[0]; pdfBlob.setName(`Application_${sanitizedFolderName}_${dateStamp}.pdf`); businessSubFolder.createFile(pdfBlob); // 8. Decode Base64 Bank Statement files and store in the same Subfolder if (data.files && Array.isArray(data.files)) { data.files.forEach(fileObj => { if (fileObj.base64 && fileObj.name) { const base64Data = fileObj.base64.split(',')[1] || fileObj.base64; const fileBlob = Utilities.newBlob( Utilities.base64Decode(base64Data), fileObj.mimeType || 'application/pdf', fileObj.name ); businessSubFolder.createFile(fileBlob); } }); } // Return Success Response return ContentService .createTextOutput(JSON.stringify({ status: "success", folderUrl: businessSubFolder.getUrl() })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
